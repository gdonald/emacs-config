#+TITLE: Emacs Configuration
#+PROPERTY: header-args:emacs-lisp :tangle init.el

Edit this file, not =init.el= or =early-init.el=. Saving it in Emacs writes both files with =org-babel-tangle=. Blocks marked =:tangle no= are not written out.

* This config

=gd-config-directory= is the repository directory, found by following the =~/.emacs.d/init.el= symlink.

#+begin_src emacs-lisp
(defconst gd-config-directory
  (file-name-directory (file-truename (or load-file-name buffer-file-name))))

(defconst gd-config-org-file (expand-file-name "config.org" gd-config-directory))

(defun gd-tangle-config-on-save ()
  "Write init.el and early-init.el when config.org is saved."
  (when (and buffer-file-name
             (string= (file-truename buffer-file-name) gd-config-org-file))
    (org-babel-tangle)))

(add-hook 'org-mode-hook
          (lambda () (add-hook 'after-save-hook #'gd-tangle-config-on-save nil t)))
#+end_src

#+begin_src emacs-lisp
(defun yarn-run-test()
  "Run a javascript test with yarn."
  (interactive)
  (let* ((file (buffer-file-name))
         (current-line (thing-at-point 'line))
         (quot nil)
         (desc nil))
    (string-match "\\(it\\|describe\\)(\\('\\|\"\\)\\(.*\\)\\('\\|\"\\)" current-line)
    (setq quot (match-string 2 current-line))
    (setq desc (match-string 3 current-line))
    (if (not (null desc))
        (progn
          (let* ((yt "*yarn test*"))
            (get-buffer-create yt)
            (switch-to-buffer yt)
            (let* ((default-directory (substring (shell-command-to-string "git rev-parse --show-toplevel") 0 -1))
                   (cmd (concat "yarn test --no-color " file " -t " quot desc quot)))
              (insert (concat "\n" cmd "\n" (make-string (length cmd) ?-) "\n\n"))
              (start-process-shell-command "yarn" yt cmd)))))))
#+end_src

** C

=c-mode= is remapped to =c-ts-mode=. The =linux= style aligns =case= with =switch=, so this style function adds a rule that indents =case= one level.

#+begin_src emacs-lisp
(defun gd-c-ts-linux-style-with-indented-case ()
  "Linux indent rules for c-ts-mode with case labels indented one level."
  (let* ((language (if (derived-mode-p 'c++-ts-mode) 'cpp 'c))
         (rules (alist-get language (c-ts-mode--simple-indent-rules language 'linux))))
    `((,language ((node-is "case") standalone-parent c-ts-mode-indent-offset)
                 ,@rules))))

(use-package c-ts-mode
  :defer t
  :custom
  (c-ts-mode-indent-offset 2)
  (c-ts-mode-indent-style #'gd-c-ts-linux-style-with-indented-case))
#+end_src
